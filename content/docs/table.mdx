Use `DataTable` for interactive tables with toolbar filters, sort dropdowns, pagination, and row selection. Low-level layout primitives live in `Table`, `TableHeader`, `TableRow`, and related components.

## Multi Select

Filter, sort, paginate, and select multiple rows with checkboxes. The header checkbox selects all rows on the current page.

```tsx
"use client";

import * as React from "react";
import { DataTable, type ColumnDef } from "@/components/ui/data-table";
import { Avatar, AvatarFallback, AvatarImage } from "@/components/ui/avatar";
import { Badge } from "@/components/ui/badge";

const columns: ColumnDef<User>[] = [
  {
    accessorKey: "name",
    header: "Name",
    cell: ({ row }) => (
      <div className="flex items-center gap-2">
        <Avatar size="sm">
          <AvatarImage src={row.avatar} alt={row.name} />
          <AvatarFallback>{row.name.slice(0, 2)}</AvatarFallback>
        </Avatar>
        <span>{row.name}</span>
      </div>
    ),
  },
  { accessorKey: "email", header: "Email" },
  { accessorKey: "role", header: "Role" },
  {
    accessorKey: "status",
    header: "Status",
    cell: ({ value }) => (
      <Badge variant={value === "active" ? "green" : "gray"}>
        {String(value)}
      </Badge>
    ),
  },
];

export function TableDemo() {
  const [selected, setSelected] = React.useState<string[]>([]);

  return (
    <DataTable
      columns={columns}
      data={users}
      getRowId={(row) => row.id}
      toolbar={{
        sort: true,
        filters: [
          { id: "search", type: "text", placeholder: "Search users…" },
          {
            id: "status", type: "select", label: "Status",
            accessorKey: "status",
            options: [
              { label: "All", value: "" },
              { label: "Active", value: "active" },
              { label: "Inactive", value: "inactive" },
            ],
          },
        ],
      }}
      pagination={{ pageSize: 8 }}
      selection={{
        mode: "multiple",
        value: selected,
        onChange: setSelected,
      }}
    />
  );
}
```
