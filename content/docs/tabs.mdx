import { Tabs, TabsList, TabsTrigger, TabsContent } from "@/components/ui/tabs";
import { Tabs as FmTabs, Tab as FmTab } from "fumadocs-ui/components/tabs";

A set of layered sections of content—known as tab panels—that are displayed one at a time. Built with `@base-ui/react`.
