title: Select

## Invalid State

Use `aria-invalid="true"` on `SelectTrigger` and `data-invalid` on `Field` to show the error state.

<Tabs items={['Preview', 'Code']} defaultIndex={0}>
  <Tab>
    <DocsPreview>
      <div className="w-full max-w-60">
        <Field data-invalid>
          <FieldLabel>Framework</FieldLabel>
          <Select>
            <SelectTrigger aria-invalid="true">
              <SelectValue placeholder="Select a framework" />
            </SelectTrigger>
            <SelectContent>
              <SelectItem value="react">React</SelectItem>
              <SelectItem value="vue">Vue</SelectItem>
              <SelectItem value="angular">Angular</SelectItem>
              <SelectItem value="svelte">Svelte</SelectItem>
            </SelectContent>
          </Select>
          <FieldError>Please select a framework.</FieldError>
        </Field>
      </div>
    </DocsPreview>
  </Tab>
  <Tab>
    <div className="flex flex-col gap-4">
      <div className="bg-fd-muted/30 p-4 rounded-lg text-sm font-mono">
